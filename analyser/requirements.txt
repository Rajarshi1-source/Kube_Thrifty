# KubeThrifty analyser -- pinned on purpose, exact minors, no ranges.
#
# An unpinned `pip install pandas` in CI is a time bomb: pandas 3.0 changed behaviour versus 2.x,
# and this repo's whole thesis is that you pin what you depend on. Renovate proposes bumps; a human
# reads the eval-gate diff before merging one.
#
# Python 3.14. Verified: every pin below resolves to a cp314 wheel -- no source builds.

pandas==3.0.2

# --- metric collection ------------------------------------------------------------------------
httpx==0.28.1                 # async Prometheus client; timeout is ALWAYS passed explicitly
prometheus-client==0.26.0     # the cgroup-truth collector's exposition format

# --- persistence ------------------------------------------------------------------------------
SQLAlchemy==2.0.52
psycopg[binary]==3.3.4        # psycopg 3; TimescaleDB is PostgreSQL 18 under the hood
redis==8.1.0                  # speaks to Valkey 9.1: streams, consumer groups, SET NX locks

# --- cluster + git ----------------------------------------------------------------------------
kubernetes==36.0.3            # 36.x tracks the 1.36 API; needed for the pods/resize subresource

# --- resilience -------------------------------------------------------------------------------
circuitbreaker==2.1.3
